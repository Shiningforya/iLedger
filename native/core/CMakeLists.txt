cmake_minimum_required(VERSION 3.22)
project(iledger_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(ILEDGER_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../.deps" CACHE PATH "Verified native dependencies")
if(NOT EXISTS "${ILEDGER_DEPS}/json.hpp")
  message(FATAL_ERROR "Run the native dependency setup before building")
endif()
add_library(iledger_core SHARED src/core.cpp src/store.cpp)
target_compile_definitions(iledger_core PRIVATE ILEDGER_CORE_BUILD)
target_include_directories(iledger_core PUBLIC include PRIVATE src "${ILEDGER_DEPS}")
if(APPLE)
  find_package(SQLite3 REQUIRED)
  target_link_libraries(iledger_core PRIVATE SQLite::SQLite3)
else()
  add_library(iledger_sqlite STATIC "${ILEDGER_DEPS}/sqlite3.c")
  target_include_directories(iledger_sqlite PUBLIC "${ILEDGER_DEPS}")
  target_compile_definitions(iledger_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
  target_link_libraries(iledger_core PRIVATE iledger_sqlite)
endif()
if(ANDROID)
  target_sources(iledger_core PRIVATE src/android_jni.cpp)
  target_link_options(iledger_core PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(iledger-core src/cli.cpp)
  target_link_libraries(iledger-core PRIVATE iledger_core)
endif()
